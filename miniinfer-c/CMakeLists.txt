cmake_minimum_required(VERSION 3.18)
project(MiniInferWorkspace LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(MINIINFER_WITH_CUDA "Enable CUDA C Runtime transfers in MiniInfer Storage" OFF)
option(MINIINFER_BUILD_NANOVLLM "Build the standalone nano-vLLM product" ON)
option(MINIINFER_BUILD_STORAGE "Build the MiniInfer storage product" ON)
option(MINIINFER_BUILD_INTEGRATION "Build the integrated miniinfer CLI" ON)
# 集成 CLI 同时依赖存储和 nano-vLLM。
if(MINIINFER_BUILD_INTEGRATION AND (NOT MINIINFER_BUILD_STORAGE OR NOT MINIINFER_BUILD_NANOVLLM))
    message(FATAL_ERROR "The integrated CLI requires both products")
endif()

enable_testing()
option(MINIINFER_WITH_KVFABRIC "Build remote storage against the installed KV-Fabric C SDK" OFF)
if(MINIINFER_BUILD_STORAGE)
    add_subdirectory(products/miniinfer-storage)
endif()
if(MINIINFER_BUILD_NANOVLLM)
    option(NANO_WITH_CUDA "Enable the standalone nano-vLLM CUDA inference kernels" OFF)
    if(NANO_WITH_CUDA)
        enable_language(CUDA)
    endif()
    add_subdirectory(products/nanovllm)
endif()
if(MINIINFER_BUILD_INTEGRATION)
    add_subdirectory(integrations/miniinfer-cli)
endif()
