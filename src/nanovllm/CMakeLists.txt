option(MINIINFER_NANO_OPENMP "Use OpenMP in nano-vLLM CPU kernels" OFF)
option(MINIINFER_NANO_BLAS "Use OpenBLAS for CPU matrix-vector products when available" ON)

set(NANO_SOURCES cache.c engine.c model.c sampler.c tokenizer.c)
add_library(miniinfer_nano STATIC ${NANO_SOURCES})
target_include_directories(miniinfer_nano PUBLIC "${PROJECT_SOURCE_DIR}/include/nanovllm")
target_compile_features(miniinfer_nano PUBLIC c_std_11)
target_compile_options(miniinfer_nano PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic>)
target_link_libraries(miniinfer_nano PUBLIC m)

if(TARGET miniinfer_remote)
    target_sources(miniinfer_nano PRIVATE nano_remote_kv.c)
    target_compile_definitions(miniinfer_nano PUBLIC NANO_WITH_REMOTE_KV=1)
    target_link_libraries(miniinfer_nano PUBLIC miniinfer_remote PRIVATE OpenSSL::Crypto)
endif()

if(MINIINFER_NANO_BLAS)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(OPENBLAS QUIET IMPORTED_TARGET openblas)
    endif()
    if(TARGET PkgConfig::OPENBLAS)
        target_compile_definitions(miniinfer_nano PRIVATE NV_USE_OPENBLAS=1)
        target_link_libraries(miniinfer_nano PUBLIC PkgConfig::OPENBLAS)
    endif()
endif()
if(MINIINFER_NANO_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS C)
    target_link_libraries(miniinfer_nano PUBLIC OpenMP::OpenMP_C)
endif()
if(NANO_WITH_CUDA)
    target_sources(miniinfer_nano PRIVATE nano_cuda.cu)
    target_compile_definitions(miniinfer_nano PUBLIC NANO_WITH_CUDA=1)
    set_target_properties(miniinfer_nano PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()

add_executable(nanovllm main.c cli.c)
target_link_libraries(nanovllm PRIVATE miniinfer_nano)

if(TARGET miniinfer_remote AND MINIINFER_BUILD_TESTS)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_executable(test_nano_remote_worker "${PROJECT_SOURCE_DIR}/tests/nanovllm/remote_worker.c")
        target_include_directories(test_nano_remote_worker PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
        target_link_libraries(test_nano_remote_worker PRIVATE miniinfer_nano)
        target_compile_options(test_nano_remote_worker PRIVATE -Wall -Wextra -Wpedantic)
        set(NANO_REMOTE_TEST
            "${Python3_EXECUTABLE}" "${PROJECT_SOURCE_DIR}/tests/nanovllm/test_remote_inference.py"
            --directory $<TARGET_FILE:kvfabricd> --worker $<TARGET_FILE:test_nano_remote_worker>
            --cli $<TARGET_FILE:nanovllm>)
        if(MINIINFER_BUILD_INTEGRATION)
            list(APPEND NANO_REMOTE_TEST --cli $<TARGET_FILE:miniinfer>)
        endif()
        if(NOT KVF_HAS_RSOCKETS)
            list(APPEND NANO_REMOTE_TEST --rdma-unavailable)
        endif()
        add_test(NAME nano_remote_inference_cpu COMMAND ${NANO_REMOTE_TEST} --device cpu)
        set_tests_properties(nano_remote_inference_cpu PROPERTIES TIMEOUT 180)
        if(NANO_WITH_CUDA)
            find_package(CUDAToolkit REQUIRED)
            target_link_libraries(test_nano_remote_worker PRIVATE CUDA::cudart)
            add_test(NAME nano_remote_inference_cuda COMMAND ${NANO_REMOTE_TEST} --device cuda)
            set_tests_properties(nano_remote_inference_cuda PROPERTIES TIMEOUT 300 SKIP_RETURN_CODE 77)
        endif()
    endif()
endif()

install(TARGETS nanovllm miniinfer_nano RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${PROJECT_SOURCE_DIR}/include/nanovllm/nv.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
