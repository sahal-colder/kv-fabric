option(MINIINFER_NANO_OPENMP "Use OpenMP in nano-vLLM CPU kernels" OFF)
option(MINIINFER_NANO_BLAS "Use OpenBLAS for CPU matrix-vector products when available" ON)

set(NANO_SOURCES cache.c engine.c model.c sampler.c tokenizer.c)
add_library(miniinfer_nano STATIC ${NANO_SOURCES})
target_include_directories(miniinfer_nano PUBLIC "${PROJECT_SOURCE_DIR}/include/nanovllm")
target_compile_features(miniinfer_nano PUBLIC c_std_11)
target_compile_options(miniinfer_nano PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall;-Wextra;-Wpedantic>)
target_link_libraries(miniinfer_nano PUBLIC m)

if(MINIINFER_NANO_BLAS)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(OPENBLAS QUIET IMPORTED_TARGET openblas)
    endif()
    if(TARGET PkgConfig::OPENBLAS)
        target_compile_definitions(miniinfer_nano PRIVATE NV_USE_OPENBLAS=1)
        target_link_libraries(miniinfer_nano PUBLIC PkgConfig::OPENBLAS)
    endif()
endif()
if(MINIINFER_NANO_OPENMP)
    find_package(OpenMP REQUIRED COMPONENTS C)
    target_link_libraries(miniinfer_nano PUBLIC OpenMP::OpenMP_C)
endif()
if(NANO_WITH_CUDA)
    target_sources(miniinfer_nano PRIVATE nano_cuda.cu)
    target_compile_definitions(miniinfer_nano PUBLIC NANO_WITH_CUDA=1)
    set_target_properties(miniinfer_nano PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
endif()

add_executable(nanovllm main.c cli.c)
target_link_libraries(nanovllm PRIVATE miniinfer_nano)
install(TARGETS nanovllm miniinfer_nano RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${PROJECT_SOURCE_DIR}/include/nanovllm/nv.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")
