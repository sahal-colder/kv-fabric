cmake_minimum_required(VERSION 3.18)
project(KVFabric LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include(GNUInstallDirs)
enable_testing()

option(MINIINFER_WITH_KVFABRIC "Build the directory and distributed Worker" ON)
option(MINIINFER_BUILD_STORAGE "Build the MiniInfer storage library" ON)
option(MINIINFER_BUILD_NANOVLLM "Build the standalone nano-vLLM engine" ON)
option(MINIINFER_BUILD_INTEGRATION "Build the MiniInfer inference CLI" ON)
option(KVF_SANITIZE "Enable address, undefined behavior and leak sanitizers" OFF)
option(MINIINFER_SANITIZE "Enable the same sanitizer build as KVF_SANITIZE" OFF)

if(MINIINFER_BUILD_INTEGRATION AND (NOT MINIINFER_BUILD_STORAGE OR NOT MINIINFER_BUILD_NANOVLLM))
    message(FATAL_ERROR "The MiniInfer CLI requires both the storage library and nano-vLLM")
endif()

if(KVF_SANITIZE OR MINIINFER_SANITIZE)
    add_compile_options(
        "$<$<COMPILE_LANGUAGE:C>:-fsanitize=address,undefined;-fno-omit-frame-pointer;-fno-pie>"
        "$<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-fsanitize=address;-Xcompiler=-fsanitize=undefined;-Xcompiler=-fno-omit-frame-pointer;-Xcompiler=-fno-pie>")
    add_link_options(-fsanitize=address,undefined -no-pie)
endif()

if(MINIINFER_WITH_KVFABRIC OR MINIINFER_BUILD_STORAGE)
    find_package(Threads REQUIRED)
endif()
if(MINIINFER_WITH_KVFABRIC)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    add_subdirectory(src/control)
endif()
if(MINIINFER_BUILD_STORAGE)
    add_subdirectory(src/worker)
endif()
if(MINIINFER_BUILD_NANOVLLM)
    option(NANO_WITH_CUDA "Enable the optional nano-vLLM CUDA inference kernels" OFF)
    if(NANO_WITH_CUDA)
        enable_language(CUDA)
    endif()
    add_subdirectory(src/nanovllm)
endif()
if(MINIINFER_BUILD_INTEGRATION)
    add_subdirectory(src/integration)
endif()
